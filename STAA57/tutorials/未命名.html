d=ABE %>% mutate(group_ind = sample(c("train","test"), 
                                    size=nrow(ABE),
                                    prob=c(0.6,0.4),
                                    replace=T)) 
m=lm(SCORE~CONFIRMED_STOREYS,data=d %>% filter(group_ind=="train"))
y.hat=predict(m,newdata=d %>% filter(group_ind =="test")) 
mean((d$SCORE[d$group_ind =="test"] - y.hat) ^ 2)
