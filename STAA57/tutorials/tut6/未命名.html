sample_die = function(n){
  s =sample(c(1, 2, 3, 4, 5, 6), size = n, replace = T)
  return(mean(s))
}
X_bar = replicate(10000, sample_die(n = 2))
plot(density(X_bar))

X_bar = replicate(10000, sample_die(n = 4))
plot(density(X_bar))

X_bar = replicate(10000, sample_die(n = 8))
plot(density(X_bar))

with the increasing value of n, the graph becomes more likely to normal distribution. if the n is big enough, the graph becomes normal distribution
